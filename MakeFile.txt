CXX      := g++
CXXFLAGS := -std=c++17 -O2 -Wall -Wextra
LDLIBS   := -lws2_32

BUILD_DIR := build
EXE := game.exe

SRC := main.cpp table.cpp console_ui.cpp net_client.cpp server.cpp game_modes.cpp

OBJ := $(addprefix $(BUILD_DIR)/,$(notdir $(SRC:.cpp=.o)))

.PHONY: all clean

all: $(EXE)

$(BUILD_DIR):
	@mkdir -p $(BUILD_DIR)

$(EXE): $(BUILD_DIR) $(OBJ)
	$(CXX) $(CXXFLAGS) $(OBJ) -o $@ $(LDLIBS)

$(BUILD_DIR)/%.o: %.cpp | $(BUILD_DIR)
	$(CXX) $(CXXFLAGS) -c $< -o $@

clean:
	@rm -rf $(BUILD_DIR) *.exe
